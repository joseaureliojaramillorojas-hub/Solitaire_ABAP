  METHOD zif_26_solitaire_spiel_jj~ist_tableau_zug_gueltig.
    DATA von_karte         TYPE REF TO zcl_26_solitaire_karte_jj.
    DATA auf_stapel_laenge TYPE i.
    DATA von_wert          TYPE i.
    DATA auf_karte         TYPE REF TO zcl_26_solitaire_karte_jj.
    DATA von_farbe         TYPE string.
    DATA zu_farbe          TYPE string.
    DATA zu_wert           TYPE i.

    "unterste Karte des Block des man bewegen will.
    von_karte = iv_karten_block->get_karte( 1 ).

    auf_stapel_laenge = iv_auf_stapel->get_size( ).

    IF auf_stapel_laenge = 0.
      von_wert = von_karte->get_wert( ).
      IF von_wert = 13.
        rv_gueltig = abap_true.
      ELSE.
        rv_gueltig = abap_false.
      ENDIF.
      RETURN.
    ENDIF.

    auf_karte = iv_auf_stapel->get_karte( auf_stapel_laenge ).

    von_farbe = von_karte->get_kartenfarbe( ).
    zu_farbe = auf_karte->get_kartenfarbe( ).
    von_wert = von_karte->get_wert( ).
    zu_wert = auf_karte->get_wert( ).
    IF von_farbe <> zu_farbe.
      IF von_wert = zu_wert - 1.
        rv_gueltig = abap_true.
      ELSE.
        rv_gueltig = abap_false.
      ENDIF.
    ELSE.
      rv_gueltig = abap_false.
    ENDIF.

      DATA(total) = von_stapel->get_size( ).
      DATA(verschobene_karten) = karten_block->remove_cards_from( 1 ).
      auf_stapel->add_cards( verschobene_karten ).
      " TODO: variable is assigned but never used (ABAP cleaner)
      DATA(entfernte_karten) = von_stapel->remove_cards_from( total - anzahl + 1 ).
      DATA(karte) = von_stapel->get_oberste_karte( ).
      karte->drehe_um( ).

    ENDIF.
